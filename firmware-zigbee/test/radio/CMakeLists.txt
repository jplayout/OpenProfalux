cmake_minimum_required(VERSION 3.16)
project(pfx_radio_host_tests C)
option(PFX_SANITIZE "Enable AddressSanitizer and UBSan" ON)
set(PORT "${CMAKE_CURRENT_LIST_DIR}/../../components/pfx_radio")
set(LEGACY "${CMAKE_CURRENT_LIST_DIR}/../../../firmware/main")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../tools/extract-radio-config.py"
    "${LEGACY}/cc1101.c" "${CMAKE_CURRENT_BINARY_DIR}/radio_profile_generated.h" COMMAND_ERROR_IS_FATAL ANY)
add_executable(pfx_radio_tests test_radio.c "${PORT}/gateway.c" "${PORT}/cc1101_port.c" "${LEGACY}/rx_confirm.c")
target_include_directories(pfx_radio_tests PRIVATE "${PORT}/include" "${LEGACY}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(pfx_radio_tests PRIVATE -std=gnu11 -Wall -Wextra -Werror -g)
if(PFX_SANITIZE)
    target_compile_options(pfx_radio_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pfx_radio_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME pfx_radio COMMAND pfx_radio_tests)
