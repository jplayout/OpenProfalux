cmake_minimum_required(VERSION 3.16)
project(pfx_usb_console_tests C)
set(PORT "${CMAKE_CURRENT_LIST_DIR}/../../components/pfx_radio")
add_executable(usb_console_tests test_usb_console.c "${PORT}/usb_console.c")
target_include_directories(usb_console_tests PRIVATE mocks "${PORT}/include")
target_compile_definitions(usb_console_tests PRIVATE setvbuf=test_setvbuf fileno=test_fileno fcntl=test_fcntl)
target_compile_options(usb_console_tests PRIVATE -std=gnu11 -Wall -Wextra -Werror -fsanitize=address,undefined -fno-omit-frame-pointer -g)
target_link_options(usb_console_tests PRIVATE -fsanitize=address,undefined)
enable_testing()
foreach(scenario IN ITEMS install reuse install-error buffer-error flags-error)
    add_test(NAME usb_${scenario} COMMAND usb_console_tests ${scenario})
endforeach()
